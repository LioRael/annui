## Installation

```bash
npm install @base-ui-components/react
```

## Usage

```tsx
import { Button } from "@base-ui-components/react";
import { PlusIcon } from "lucide-react";
```

```tsx
<Button.Root>
  <Button.Icon render={<PlusIcon />} />
  Click me
</Button.Root>
```
